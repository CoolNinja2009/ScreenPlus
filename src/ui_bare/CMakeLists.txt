cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ScreenPlusBarePreview LANGUAGES C)
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_C_STANDARD_REQUIRED ON)
endif()
# Bare: a checkout beside this one (Projects/Bare), else Bare at BARE_COMMIT from GitHub (CI). Move BARE_COMMIT on
# when Screen+ needs newer Bare.
set(BARE_COMMIT e8ee0153bc76ae2e3aa8aeade5d9ccb96d8bdf25)
set(BARE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../Bare" CACHE PATH "Bare source checkout")
set(BARE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BARE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BARE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
if(NOT TARGET bare)
    if(EXISTS "${BARE_SOURCE_DIR}/CMakeLists.txt")
        add_subdirectory("${BARE_SOURCE_DIR}" bare-framework)
    else()
        message(STATUS "No Bare at ${BARE_SOURCE_DIR}: fetching Bare ${BARE_COMMIT}")
        include(FetchContent)
        FetchContent_Declare(bare URL https://github.com/CoolNinja2009/Bare/archive/${BARE_COMMIT}.tar.gz)
        FetchContent_MakeAvailable(bare)
    endif()
endif()
set(SP_VIEW screenplus_bare.c sp_kit.c ../common/json.c ../common/util.c)
if(NOT WIN32)
    list(APPEND SP_VIEW ../linux/compat.c ../linux/winobj.c)
    add_compile_definitions(_GNU_SOURCE)   # compat.c, as in src/linux
    # the Windows API shims take parameters they have no use for (src/linux builds them with -Wno-unused too)
    set_source_files_properties(../linux/compat.c ../linux/winobj.c PROPERTIES COMPILE_OPTIONS "-Wno-unused")
endif()
add_executable(screenplus-bare-preview preview.c ${SP_VIEW})
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set_target_properties(screenplus-bare-preview PROPERTIES EXCLUDE_FROM_ALL ON)   # a design tool, not shipped
endif()
target_link_libraries(screenplus-bare-preview PRIVATE bare)
add_executable(screenplus-bare-ui WIN32 live.c sp_ipc.c ${SP_VIEW})
target_link_libraries(screenplus-bare-ui PRIVATE bare)
set_target_properties(screenplus-bare-ui PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
# The graphs' shader (shaders/graph.glsl): compiled into generated/sp_shaders.h when glslc and shadercross are
# here, else the committed header is used (commit it after changing the shader).
bare_add_shaders(screenplus-bare-ui OUTPUT ${CMAKE_CURRENT_LIST_DIR}/generated/sp_shaders.h
                 SOURCES ${CMAKE_CURRENT_LIST_DIR}/shaders/graph.glsl)
target_include_directories(screenplus-bare-preview PRIVATE ${CMAKE_CURRENT_LIST_DIR}/generated)
if(TARGET screenplus-bare-ui_shaders)
    add_dependencies(screenplus-bare-preview screenplus-bare-ui_shaders)
endif()
foreach(t screenplus-bare-ui screenplus-bare-preview)
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /utf-8)
        target_compile_definitions(${t} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${t} PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
    endif()
endforeach()
if(WIN32)
    target_sources(screenplus-bare-ui PRIVATE screenplus_bare.rc)
    foreach(t screenplus-bare-ui screenplus-bare-preview)
        target_link_libraries(${t} PRIVATE shell32 shlwapi advapi32 bcrypt user32)
    endforeach()
    if(MSVC)
        target_link_options(screenplus-bare-ui PRIVATE /ENTRY:mainCRTStartup)
    else()
        target_link_options(screenplus-bare-ui PRIVATE -mwindows)
    endif()
else()
    target_link_libraries(screenplus-bare-ui PRIVATE m pthread)
    target_link_libraries(screenplus-bare-preview PRIVATE m)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(BUILD_TESTING AND CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(screenplus-bare-view-test test_view.c ${SP_VIEW})
    target_link_libraries(screenplus-bare-view-test PRIVATE bare)
    target_include_directories(screenplus-bare-view-test PRIVATE ${CMAKE_CURRENT_LIST_DIR}/generated)
    if(WIN32)
        target_link_libraries(screenplus-bare-view-test PRIVATE bcrypt)
    endif()
    add_test(NAME screenplus-bare-view COMMAND screenplus-bare-view-test)
endif()
