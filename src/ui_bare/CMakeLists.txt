# Bare: a checkout beside this one (Projects/Bare), else Bare at BARE_COMMIT from GitHub (CI). Move BARE_COMMIT on

    if(EXISTS "${BARE_SOURCE_DIR}/CMakeLists.txt")
        add_subdirectory("${BARE_SOURCE_DIR}" bare-framework)
    else()
        message(STATUS "No Bare at ${BARE_SOURCE_DIR}: fetching Bare ${BARE_COMMIT}")
        include(FetchContent)
        FetchContent_Declare(bare URL https://github.com/CoolNinja2009/Bare/archive/${BARE_COMMIT}.tar.gz)
        FetchContent_MakeAvailable(bare)
    endif()

if(NOT WIN32)
    list(APPEND SP_VIEW ../linux/compat.c ../linux/winobj.c)
    add_compile_definitions(_GNU_SOURCE)   # compat.c, as in src/linux
    # the Windows API shims take parameters they have no use for (src/linux builds them with -Wno-unused too)
    set_source_files_properties(../linux/compat.c ../linux/winobj.c PROPERTIES COMPILE_OPTIONS "-Wno-unused")

# here, else the committed header is used (commit it after changing the shader).

                 SOURCES ${CMAKE_CURRENT_LIST_DIR}/shaders/graph.glsl)

if(TARGET screenplus-bare-ui_shaders)
    add_dependencies(screenplus-bare-preview screenplus-bare-ui_shaders)
endif()
foreach(t screenplus-bare-ui screenplus-bare-preview)
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /utf-8)
        target_compile_definitions(${t} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${t} PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
    endif()
endforeach()
if(WIN32)
    target_sources(screenplus-bare-ui PRIVATE screenplus_bare.rc)
    foreach(t screenplus-bare-ui screenplus-bare-preview)
        target_link_libraries(${t} PRIVATE shell32 shlwapi advapi32 bcrypt user32)
    endforeach()
    if(MSVC)
        target_link_options(screenplus-bare-ui PRIVATE /ENTRY:mainCRTStartup)
    else()
        target_link_options(screenplus-bare-ui PRIVATE -mwindows)
    endif()
else()
    target_link_libraries(screenplus-bare-ui PRIVATE m pthread)
    target_link_libraries(screenplus-bare-preview PRIVATE m)
endif()
# Sample app wallpapers (Windows): Bare programs from Bare's examples/, each packed into <build>/<Name>.spf, the file
# a release carries (CI uploads *.spf). Their shaders come from the headers committed in Bare (examples/<dir>/generated).
if(WIN32 AND NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(DEFINED bare_SOURCE_DIR)
        set(_bare_root "${bare_SOURCE_DIR}")
    else()
        set(_bare_root "${BARE_SOURCE_DIR}")
    endif()
    function(screenplus_sample_app name dir)   # [title]: the library's, when it isn't the file name
        set(title "${name}")
        if(ARGC GREATER 2)
            set(title "${ARGV2}")
        endif()
        set(src "${_bare_root}/examples/${dir}")
        add_executable(sample-${dir} WIN32 "${src}/${dir}.c")
        target_link_libraries(sample-${dir} PRIVATE bare)
        target_include_directories(sample-${dir} PRIVATE "${src}/generated")
        set_target_properties(sample-${dir} PROPERTIES OUTPUT_NAME ${dir}
                              RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/sample-apps/${dir}")
        if(MSVC)
            target_compile_options(sample-${dir} PRIVATE /W4 /utf-8)
            target_compile_definitions(sample-${dir} PRIVATE _CRT_SECURE_NO_WARNINGS)
            target_link_options(sample-${dir} PRIVATE /ENTRY:mainCRTStartup)
        else()
            target_link_options(sample-${dir} PRIVATE -mwindows)
        endif()
        add_custom_command(TARGET sample-${dir} POST_BUILD
            COMMAND "${CMAKE_COMMAND}" "-DNAME=${name}" "-DTITLE=${title}" "-DEXE=$<TARGET_FILE:sample-${dir}>"
                    "-DPROPS=${src}/LivelyProperties.json" "-DPREVIEW=${src}/preview.jpg" "-DOUT=${CMAKE_BINARY_DIR}/${name}.spf"
                    -P "${CMAKE_SOURCE_DIR}/cmake/pack_spf.cmake"
            VERBATIM)
    endfunction()
    screenplus_sample_app(Aurora aurora)
    screenplus_sample_app(Singularity singularity)
    screenplus_sample_app(Fluids fluid)
    screenplus_sample_app(KoiPond koi "Koi Pond")
endif()
